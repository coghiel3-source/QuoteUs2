### Import via `psql` (fastest)

From the server where Postgres is running:

1. Save the CSV as `vehicles.csv`.
2. Run:

```bash
psql $DATABASE_URL -c "\copy vehicles(year,make,model,trim) FROM 'vehicles.csv' WITH (FORMAT csv, HEADER true)"
```

(Replace `$DATABASE_URL` or use `psql -h host -U user -d dbname` accordingly.)

### Node.js seed script (bulk import from CSV using `pg` and `csv-parse`)

Create `scripts/seed_vehicles.js`:

```js
// scripts/seed_vehicles.js
const fs = require('fs');
const { parse } = require('csv-parse');
const { Pool } = require('pg');

const pool = new Pool({ connectionString: process.env.DATABASE_URL });

async function seed() {
  const parser = fs.createReadStream('vehicles.csv').pipe(parse({ columns: true, trim: true }));
  const client = await pool.connect();
  try {
    await client.query('BEGIN');
    for await (const row of parser) {
      const { year, make, model, trim } = row;
      await client.query(
        `INSERT INTO vehicles (year, make, model, trim) VALUES ($1, $2, $3, $4) ON CONFLICT DO NOTHING`,
        [parseInt(year, 10), make, model, trim || null]
      );
    }
    await client.query('COMMIT');
    console.log('Vehicles seeded');
  } catch (err) {
    await client.query('ROLLBACK');
    console.error('Seed failed', err);
  } finally {
    client.release();
    pool.end();
  }
}

seed();
```

Run:

```bash
DATABASE_URL=postgres://user:pass@host:5432/dbname node scripts/seed_vehicles.js
```

### Notes & best practices

* If you have a large official vehicle dataset (recommended), import via `COPY` for speed.
* Keep a `source` column if you import from multiple vendors. Add `trim`/`submodel` columns if needed.
* Provide an API endpoint `GET /api/vehicles?year=&make=` that queries distinct makes/models for dependent selects (use indexed queries).

#### `src/server/routes/vehicles.ts` (example)

```ts
import { Router } from 'express';
import { Pool } from 'pg';
const router = Router();
const pool = new Pool({ connectionString: process.env.DATABASE_URL });

router.get('/', async (req, res) => {
  const { year, make } = req.query;
  try {
    if (year && make) {
      const r = await pool.query('SELECT DISTINCT model FROM vehicles WHERE year=$1 AND make=$2 ORDER BY model', [year, make]);
      return res.json(r.rows);
    } else if (year) {
      const r = await pool.query('SELECT DISTINCT make FROM vehicles WHERE year=$1 ORDER BY make', [year]);
      return res.json(r.rows);
    } else {
      const r = await pool.query('SELECT DISTINCT year FROM vehicles ORDER BY year DESC');
      return res.json(r.rows);
    }
  } catch (err) {
    res.status(500).json({ error: 'DB error' });
  }
});

export default router;
```

export function estimateAutoPremium(profile: any): { low:number, mid:number, high:number } {
  // simple mock: base by vehicle age and adders for young drivers, claims, km
  const base = 800 - Math.min(20 * (profile.vehicleYear ? (new Date().getFullYear() - profile.vehicleYear) : 0), 400);
  let adj = 0;
  if (profile.drivers.some((d:any)=> new Date().getFullYear() - new Date(d.licenseDate).getFullYear() < 2)) adj += 300;
  if (profile.claims && profile.claims > 0) adj += 400;
  adj += (profile.annualKm || 12000) / 1000 * 20;
  const mid = Math.max(350, base + adj);
  return { low: Math.round(mid*0.85), mid: Math.round(mid), high: Math.round(mid*1.25) };
}
